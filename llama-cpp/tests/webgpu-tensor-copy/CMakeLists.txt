cmake_minimum_required(VERSION 3.24)
project(copy_overlay_check LANGUAGES CXX)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
set(EMSCRIPTEN TRUE)
set(GGML_WEBGPU_JSPI ON)
set(LCB_WEBGPU ON)
function(ggml_add_backend_library name)
    add_library(${name} STATIC ${ARGN})
endfunction()
add_subdirectory("${LCB_LLAMA_SOURCE}/ggml/src/ggml-webgpu" webgpu)
get_target_property(before ggml-webgpu SOURCES)
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/before.txt" "${before}")
if(LCB_WEBGPU_MOE_DIRECT_SLOT)
    include(../../cmake/MoeDirectSlotOverlay.cmake)
endif()
add_library(llama STATIC "${LCB_LLAMA_SOURCE}/src/llama-model-loader.cpp")
set_property(TARGET llama PROPERTY SOURCES "llama-model-loader.cpp")
include(../../cmake/WebgpuSourceOverlay.cmake)
get_target_property(after ggml-webgpu SOURCES)
get_target_property(includes ggml-webgpu INCLUDE_DIRECTORIES)
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/after.txt" "${after}")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/includes.txt" "${includes}")
# Real mtmd source preparation is covered independently by existing tests;
# this fixture exercises all overlay hooks against independent targets at once.
add_library(mtmd STATIC "${LCB_LLAMA_SOURCE}/tools/mtmd/clip.cpp")
set_property(TARGET mtmd PROPERTY SOURCES "clip.cpp;mtmd-audio.cpp")
include(../../cmake/MtmdOverlay.cmake)
include(../../cmake/MtmdAudioOverlay.cmake)
get_target_property(mtmd_sources mtmd SOURCES)
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/mtmd.txt" "${mtmd_sources}")
