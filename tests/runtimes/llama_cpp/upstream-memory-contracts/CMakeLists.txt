cmake_minimum_required(VERSION 3.24)
project(upstream_memory_contracts LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(GGML_CPU OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_METAL OFF CACHE BOOL "" FORCE)
set(GGML_ACCELERATE OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(GGML_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
add_subdirectory("${LCB_LLAMA_SOURCE}/ggml" ggml)
add_executable(upstream-memory-contracts contracts.cpp)
target_include_directories(upstream-memory-contracts PRIVATE "${LCB_LLAMA_SOURCE}/ggml/src")
target_link_libraries(upstream-memory-contracts PRIVATE ggml-base)
enable_testing()
add_test(NAME upstream-memory-contracts COMMAND upstream-memory-contracts)
