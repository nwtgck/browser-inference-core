cmake_minimum_required(VERSION 3.24)
project(upstream_recurrent_contracts LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(GGML_CPU ON CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_METAL OFF CACHE BOOL "" FORCE)
set(GGML_ACCELERATE OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_CPU_KLEIDIAI OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_COMMON ON CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_MTMD OFF CACHE BOOL "" FORCE)
add_subdirectory("${LCB_LLAMA_SOURCE}" llama)
add_executable(upstream-recurrent-contracts model.cpp)
target_include_directories(upstream-recurrent-contracts PRIVATE
    "${LCB_LLAMA_SOURCE}/include" "${LCB_LLAMA_SOURCE}/common"
    "${LCB_LLAMA_SOURCE}/ggml/include" "${LCB_LLAMA_SOURCE}/src" "${LCB_LLAMA_SOURCE}/tests")
target_link_libraries(upstream-recurrent-contracts PRIVATE llama-common)
enable_testing()
add_test(NAME upstream-recurrent-contracts COMMAND upstream-recurrent-contracts)
