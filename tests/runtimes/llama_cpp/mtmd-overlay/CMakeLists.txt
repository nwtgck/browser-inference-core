cmake_minimum_required(VERSION 3.24)
project(mtmd_overlay_check LANGUAGES C CXX)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
# Configure the actual upstream mtmd target; build only its overlay translation
# unit when checking this fixture, not the language models or runtime artifacts.
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
foreach(option BUILD_TESTS BUILD_TOOLS BUILD_EXAMPLES BUILD_SERVER BUILD_APP OPENSSL SUBPROCESS)
    set(LLAMA_${option} OFF CACHE BOOL "" FORCE)
endforeach()
set(LLAMA_BUILD_MTMD ON CACHE BOOL "" FORCE)
set(LLAMA_BUILD_COMMON OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_METAL OFF CACHE BOOL "" FORCE)
set(GGML_BLAS OFF CACHE BOOL "" FORCE)
set(MTMD_VIDEO OFF CACHE BOOL "" FORCE)
add_subdirectory("${LCB_LLAMA_SOURCE}" llama)
get_target_property(upstream_sources mtmd SOURCES)
include("${CMAKE_CURRENT_LIST_DIR}/snapshot-target-property.cmake")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/upstream-mtmd-sources.txt" "${upstream_sources}\n")
lcb_snapshot_target_property(mtmd PRECOMPILE_HEADERS
    "${CMAKE_CURRENT_BINARY_DIR}/upstream-mtmd-headers.txt")
if(NOT DEFINED LCB_WEBGPU)
    set(LCB_WEBGPU ON)
endif()
set(LCB_WEBGPU_BF16_PROJECTOR ON)
include(../../../../runtimes/llama-cpp/cmake/MtmdOverlay.cmake)
get_target_property(sources mtmd SOURCES)
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/mtmd-sources.txt" "${sources}\n")
lcb_snapshot_target_property(mtmd PRECOMPILE_HEADERS
    "${CMAKE_CURRENT_BINARY_DIR}/mtmd-headers.txt")

# Record source properties from the target's directory, where upstream set them.
set(property_rows "")
foreach(source IN LISTS sources)
    get_source_file_property(skip_unity "${source}" TARGET_DIRECTORY mtmd SKIP_UNITY_BUILD_INCLUSION)
    string(APPEND property_rows "${source}|${skip_unity}\n")
endforeach()
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/mtmd-source-properties.txt" "${property_rows}")
get_target_property(mtmd_unity mtmd UNITY_BUILD)
get_target_property(mtmd_unity_batch mtmd UNITY_BUILD_BATCH_SIZE)
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/mtmd-unity-properties.txt" "${mtmd_unity}|${mtmd_unity_batch}\n")
